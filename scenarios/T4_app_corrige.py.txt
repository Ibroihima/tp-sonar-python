def add(a, b):
    return a + b


def divide(a, b):
    if b == 0:
        raise ValueError("Division par zero")
    return a / b


def is_even(n):
    return n % 2 == 0


def read_config(path):
    with open(path, encoding="utf-8") as f:
        return f.read()


def classify(x):
    return "grand" if x > 10 else "petit"
